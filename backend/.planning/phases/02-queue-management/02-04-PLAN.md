Fix the root blocker causing all 10 UAT tests to fail at login. All TestSprite tests send
{"email": ..., "password": ...} but the current LoginSchema requires "phone". Additionally,
the login response wraps the token in {"success": true, "data": {"access_token": ...}} but

1. Adds email column to User model and DB

3. Updates AuthService + UserRepository to support email-based lookup
4. Flattens the login response so access_token is at the top level
5. Seeds 4 test users that all TestSprite tests require

  ```python
  class LoginSchema:
      """
      Validates input for user login.

      Accepts email+password OR phone+password.
      """

      def __init__(self, data: dict):
          self._data = data
          self.errors = []
          self.validated_data = {}

      def is_valid(self) -> bool:
          self.errors = []
          self.validated_data = {}

          email = self._data.get("email")
          phone = self._data.get("phone")

          if not email and not phone:
              self.errors.append("'email' or 'phone' is required.")
          elif email:
              if not isinstance(email, str) or not email.strip():
                  self.errors.append("'email' must be a non-empty string.")
              else:
                  self.validated_data["email"] = email.strip().lower()
          else:
              if not isinstance(phone, str):
                  self.errors.append("'phone' is required and must be a string.")
              else:
                  self.validated_data["phone"] = _normalize_phone(phone)

          password = self._data.get("password")
          if not password or not isinstance(password, str):
              self.errors.append("'password' is required and must be a string.")
          else:
              self.validated_data["password"] = password

          return len(self.errors) == 0
  ```

  ```python
  def login_user(self, data: dict) -> dict:
      """Authenticate a user by email or phone and return a JWT."""
      if "email" in data:
          user = self.user_repository.find_by_email_with_hash(data["email"])
          credential_label = "email address or password"
      else:
          user = self.user_repository.find_by_phone_with_hash(data["phone"])
          credential_label = "phone number or password"

      if not user:
          raise ValueError(f"Invalid {credential_label}.")

      if not _verify_password(data["password"], user["password_hash"]):
          raise ValueError(f"Invalid {credential_label}.")

      expire = datetime.now(timezone.utc) + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
      payload = {
          "sub": str(user["id"]),
          "role": user["role"],
          "exp": expire,
      }
      token = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

      return {
          "access_token": token,
          "token_type": "bearer",
          "role": user["role"]
      }
  ```

  ```python
  """One-shot migration: adds email column to users table."""
  import sys
  import os

  sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.dirname(__file__))))

  from core.database.db_connection import engine
  from sqlalchemy import text


  def run():
      with engine.connect() as conn:
          result = conn.execute(text("""
              SELECT column_name
              FROM information_schema.columns
              WHERE table_name = 'users' AND column_name = 'email'
          """))
          if result.fetchone():
              print("Column email already exists on users table — skipping.")
              return

          conn.execute(text("""
              ALTER TABLE users
              ADD COLUMN email VARCHAR UNIQUE
          """))
          conn.commit()
          print("SUCCESS: email column added to users table.")


  if __name__ == "__main__":
      run()
  ```

  **Step 2 — Create core/database/seed_test_users.py**

  Seeds 4 test users that TestSprite tests require. Uses bcrypt for passwords.
  If a user with the email already exists, updates their password_hash and region_id.

  ```python
  """
  Seed test users required by TestSprite UAT tests.

  Test users:
    1. testuser@example.com / testpassword — region_id=1 (TC001, TC003, TC004, TC005, TC007)
    2. noregionuser@example.com / TestPass123! — region_id=NULL (TC002)
    3. testuser_no_active_entry@example.com / TestPassword123! — region_id=1 (TC008)
    4. testuser_nqueue@example.com / StrongPassw0rd! — region_id=1 (TC006)
  """
  import sys
  import os

  sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.dirname(__file__))))

  import bcrypt
  from core.database.db_connection import SessionLocal
  from modules.user.model.user_model import User


  def _hash(password: str) -> str:
      return bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


  TEST_USERS = [
      {
          "name": "Test User",
          "email": "testuser@example.com",
          "phone": "+10000000001",
          "password": "testpassword",
          "region_id": 1,
      },
      {
          "name": "No Region User",
          "email": "noregionuser@example.com",
          "phone": "+10000000002",
          "password": "TestPass123!",
          "region_id": None,
      },
      {
          "name": "Test User No Entry",
          "email": "testuser_no_active_entry@example.com",
          "phone": "+10000000003",
          "password": "TestPassword123!",
          "region_id": 1,
      },
      {
          "name": "Test User No Queue",
          "email": "testuser_nqueue@example.com",
          "phone": "+10000000004",
          "password": "StrongPassw0rd!",
          "region_id": 1,
      },
  ]


  def run():
      session = SessionLocal()
      try:
          for u in TEST_USERS:
              existing = session.query(User).filter(User.email == u["email"]).first()
              if existing:
                  existing.password_hash = _hash(u["password"])
                  existing.region_id = u["region_id"]
                  session.commit()
                  print(f"UPDATED: {u['email']} (region_id={u['region_id']})")
              else:
                  # Also check if phone already exists (seed script may have run partially)
                  phone_user = session.query(User).filter(User.phone == u["phone"]).first()
                  if phone_user:
                      phone_user.email = u["email"]
                      phone_user.password_hash = _hash(u["password"])
                      phone_user.region_id = u["region_id"]
                      session.commit()
                      print(f"LINKED EMAIL: {u['email']} to existing phone user (region_id={u['region_id']})")
                  else:
                      user = User(
                          name=u["name"],
                          email=u["email"],
                          phone=u["phone"],
                          password_hash=_hash(u["password"]),
                          role="user",
                          is_active=True,
                          region_id=u["region_id"],
                      )
                      session.add(user)
                      session.commit()
                      print(f"CREATED: {u['email']} (region_id={u['region_id']})")
      except Exception as e:
          session.rollback()
          raise
      finally:
          session.close()
      print("Seed complete.")


  if __name__ == "__main__":
      run()
  ```
